# פתרון – תרגיל Aggregation על ציונים (למורה)

## `src/models/GradeModel.ts`
```ts
import mongoose, { model } from 'mongoose';

const gradeSchema = new mongoose.Schema({
  studentName: { type: String, required: true },
  className: { type: String, required: true },
  subject: { type: String, required: true },
  score: { type: Number, required: true },
  createdAt: { type: Date, default: Date.now }
});

export default model('Grade', gradeSchema);
```

## `src/controllers/gradeController.ts`
```ts
import { Request, Response } from 'express';
import GradeModel from '../models/GradeModel';

export async function createGrade(req: Request, res: Response): Promise<void> {
  try {
    const grade = await GradeModel.create(req.body);
    res.status(201).json(grade);
  } catch (error) {
    res.status(500).json({ message: 'Error creating grade' });
  }
}

export async function getGrades(req: Request, res: Response): Promise<void> {
  try {
    const grades = await GradeModel.find();
    res.status(200).json(grades);
  } catch (error) {
    res.status(500).json({ message: 'Error getting grades' });
  }
}

// משימה 1
export const getAverageScore = async (req: Request, res: Response) => {
  const result = await GradeModel.aggregate([
    { $group: { _id: null, averageScore: { $avg: '$score' } } }
  ]);
  res.json(result[0]);
};

// משימה 2
export const getAverageByClass = async (req: Request, res: Response) => {
  const result = await GradeModel.aggregate([
    {
      $group: {
        _id: '$className',
        averageScore: { $avg: '$score' },
        studentsCount: { $sum: 1 }
      }
    }
  ]);
  res.json(result);
};

// משימה 3
export const getPassedGrades = async (req: Request, res: Response) => {
  const result = await GradeModel.aggregate([
    { $match: { score: { $gte: 60 } } }
  ]);
  res.json(result);
};

// משימה 4
export const getTopSubjects = async (req: Request, res: Response) => {
  const result = await GradeModel.aggregate([
    { $group: { _id: '$subject', averageScore: { $avg: '$score' } } },
    { $sort: { averageScore: -1 } }
  ]);
  res.json(result);
};

// בונוס 1
export const getFailedByClass = async (req: Request, res: Response) => {
  const result = await GradeModel.aggregate([
    { $match: { score: { $lt: 60 } } },
    { $group: { _id: '$className', failedCount: { $sum: 1 } } }
  ]);
  res.json(result);
};

// בונוס 2
export const getMaxBySubject = async (req: Request, res: Response) => {
  const result = await GradeModel.aggregate([
    { $group: { _id: '$subject', maxScore: { $max: '$score' } } }
  ]);
  res.json(result);
};
```

## `src/routes/gradeRoutes.ts`
```ts
import { Router } from 'express';
import {
  createGrade, getGrades,
  getAverageScore, getAverageByClass, getPassedGrades, getTopSubjects,
  getFailedByClass, getMaxBySubject
} from '../controllers/gradeController';

const router = Router();

router.post('/', createGrade);
router.get('/', getGrades);
router.get('/average', getAverageScore);
router.get('/average-by-class', getAverageByClass);
router.get('/passed', getPassedGrades);
router.get('/top-subjects', getTopSubjects);
router.get('/failed-by-class', getFailedByClass);
router.get('/max-by-subject', getMaxBySubject);

export default router;
```

## שינוי ב-`src/app.ts`
```ts
import gradeRoutes from './routes/gradeRoutes';
// ...
app.use('/grades', gradeRoutes);
```
